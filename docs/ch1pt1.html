Separate Station.Start

Wanted to merge predator data with grid data by GridLine & GridStation

This would assign each observation to a region

# ensure Station.Start is character
predator_data <- predator_data %>%
  mutate(Station.Start = as.character(Station.Start))

# separate gridline and gridstation
predator_data <- predator_data %>%
  mutate(
    # Extract GridLine (before first dot)
    GridLine = as.numeric(str_extract(Station.Start, "^[^\\.]+")),
    # Extract the part after the first dot
    station_part = str_replace(Station.Start, "^[^\\.]+\\.", "")
  ) %>%
  mutate(
    GridStation_raw = case_when(
      # e.g. "14.5"
      str_detect(station_part, "\\.") ~ as.numeric(station_part),
      
      # e.g. "-020"
      str_detect(station_part, "^\\-\\d+$") ~ as.numeric(station_part),
      
      # three or more digits -> direct numeric (e.g. "080" -> 80)
      str_detect(station_part, "^\\d{3,}$") ~ as.numeric(station_part),
      
      # two digits -> multiply by 10 (e.g. "04" -> 40, "14" -> 140)
      str_detect(station_part, "^\\d{2}$") ~ as.numeric(station_part) * 10,
      
      # one digit -> multiply by 100 (e.g. "2" -> 200)
      str_detect(station_part, "^\\d{1}$") ~ as.numeric(station_part) * 100,
      
      TRUE ~ NA_real_
    )
  ) %>%
  mutate(
    # round to nearest 20 (to match bin size)
    GridStation = if_else(!is.na(GridStation_raw),
                          round(GridStation_raw / 20) * 20,
                          NA_real_)
  ) %>%
  select(-station_part, -GridStation_raw)

Gap statistic to find optimal number of clusters

# k-means for range of potential clusters
# wss = within-cluster sum of squared errors (inertia?)

# number of clusters to test
k_values <- 1:10
wss_values <- numeric(length(k_values))  # empty vector to keep k_avlues

# loop over k = 1 to 10
for (i in seq_along(k_values)) {
  k <- k_values[i]
# cut tree into k clusters
  groups <- cutree(hc, k = k)
# get list of unique clusters
  cluster_ids <- unique(groups)
# begin WSS for this k
  total_wss <- 0
# loop over each cluster
  for (g in cluster_ids) {
    members <- which(groups == g)                     # indices of members
    distance <- as.matrix(dist_matrix)[members, members]     # distances within cluster
    total_wss <- total_wss + sum(distance^2) / (2 * length(members))
  }
  
  # keep result
  wss_values[i] <- total_wss
}

# plot elbow curve
# the point in between where there is a steep slope is number to use
plot(k_values, wss_values, type = "b", pch = 19,
     xlab = "Number of clusters (k)",
     ylab = "Within-cluster sum of squares",
     main = "Elbow Method to Estimate Optimal Number of Clusters")
png("elbow_plot.png", width = 800, height = 600)
